
// ==UserScript==
// @name         Tennr Auto Hunt V6.3 Reliable Fax Detection
// @namespace    tennr
// @version      6.3
// @description  F8/F9 automatic hunting with reliable fax detection and auto-resume
// @match        https://app.tennr.com/*
// @grant        none
// ==/UserScript==

(function () {
    'use strict';

    const AUTO_HUNT_KEY = 'tennrAutoHuntV6';
    const STATS_KEY = 'tennrAutoHuntStatsV6';

    const START_KEY = 'F8';
    const STOP_KEY = 'F9';

    const PAGE_CHECK_INTERVAL = 250;
    const TASK_OPEN_CONFIRMATIONS = 2;
    const WORKSPACE_CONFIRMATIONS = 3;
    const RESUME_DELAY = 1000;

    let autoModeEnabled =
        localStorage.getItem(AUTO_HUNT_KEY) === 'on';

    let hunting = false;
    let taskOpen = false;

    let ctrlSpamTimer = null;
    let cycleTimeout = null;
    let holdTimeout = null;
    let reloadTimeout = null;
    let resumeTimeout = null;
    let pageMonitorTimer = null;
    let taskObserver = null;

    let taskOpenChecks = 0;
    let workspaceChecks = 0;

    let cycleStartUrl = location.href;
    let lastTaskCandidate = null;

    let stats = loadStats();

    /**
     * Load saved statistics.
     */
    function loadStats() {
        try {
            const saved = JSON.parse(
                localStorage.getItem(STATS_KEY) || '{}'
            );

            return {
                retries: Number(saved.retries) || 0,
                apiAttempts: Number(saved.apiAttempts) || 0,
                apiCandidates: Number(saved.apiCandidates) || 0,
                confirmedTasks: Number(saved.confirmedTasks) || 0
            };
        } catch (error) {
            return {
                retries: 0,
                apiAttempts: 0,
                apiCandidates: 0,
                confirmedTasks: 0
            };
        }
    }

    /**
     * Save statistics.
     */
    function saveStats() {
        try {
            localStorage.setItem(
                STATS_KEY,
                JSON.stringify(stats)
            );
        } catch (error) {
            console.debug(
                '[Tennr Hunt] Could not save statistics:',
                error
            );
        }
    }

    /**
     * Check whether an element is visible.
     */
    function isElementVisible(element) {
        if (!element || !element.isConnected) {
            return false;
        }

        const style = window.getComputedStyle(element);
        const rectangle = element.getBoundingClientRect();

        return (
            style.display !== 'none' &&
            style.visibility !== 'hidden' &&
            Number(style.opacity) !== 0 &&
            rectangle.width > 0 &&
            rectangle.height > 0
        );
    }

    /**
     * Find a visible element.
     */
    function findVisibleElement(selector) {
        try {
            return [...document.querySelectorAll(selector)]
                .find(isElementVisible) || null;
        } catch (error) {
            return null;
        }
    }

    /**
     * Find a visible button by its exact text or title.
     */
    function findVisibleButton(text) {
        const normalizedTarget = text
            .replace(/\s+/g, ' ')
            .trim()
            .toLowerCase();

        return [...document.querySelectorAll('button')]
            .find((button) => {
                if (!isElementVisible(button)) {
                    return false;
                }

                const buttonText = (
                    button.textContent || ''
                )
                    .replace(/\s+/g, ' ')
                    .trim()
                    .toLowerCase();

                const buttonTitle = (
                    button.getAttribute('title') || ''
                )
                    .replace(/\s+/g, ' ')
                    .trim()
                    .toLowerCase();

                return (
                    buttonText === normalizedTarget ||
                    buttonTitle === normalizedTarget
                );
            }) || null;
    }

    /**
     * Get normalized visible body text.
     */
    function getPageText() {
        return (
            document.body?.innerText || ''
        )
            .replace(/\s+/g, ' ')
            .trim()
            .toLowerCase();
    }

    /**
     * Detect the actual open fax/task interface.
     *
     * These selectors come from the provided open-fax HTML.
     */
    function isFaxTaskOpen() {
        /*
         * Primary and strongest signal from the provided HTML.
         */
        const sourceViewer =
            document.getElementById(
                'SourceViewerContainer'
            );

        if (
            sourceViewer &&
            isElementVisible(sourceViewer)
        ) {
            return true;
        }

        /*
         * React PDF viewer shown in an open fax.
         */
        const pdfDocument =
            findVisibleElement(
                '.react-pdf__Document'
            );

        if (pdfDocument) {
            return true;
        }

        /*
         * Individual PDF page canvases.
         */
        const pdfPage =
            findVisibleElement(
                '.react-pdf__Page'
            );

        if (pdfPage) {
            return true;
        }

        /*
         * Strong workflow action from the provided HTML.
         */
        const everythingLooksRight =
            findVisibleButton(
                'Everything looks right'
            );

        if (everythingLooksRight) {
            return true;
        }

        /*
         * Multiple secondary signals for the form area.
         */
        const validationSection =
            findVisibleElement(
                '[data-validate-scroll-target="true"]'
            );

        const validationField =
            findVisibleElement(
                '[id^="validate-entry-"]'
            );

        if (
            validationSection &&
            validationField
        ) {
            return true;
        }

        /*
         * PDF viewer filename selector.
         */
        const pdfFilename =
            findVisibleElement(
                '[title$=".pdf" i]'
            );

        if (pdfFilename) {
            return true;
        }

        /*
         * Backup combination from the open-task page.
         */
        const pageText = getPageText();

        const hasPatientSearch =
            pageText.includes('patient search');

        const hasDocumentClassification =
            pageText.includes(
                'document classification'
            );

        const hasBadFax =
            pageText.includes('bad fax');

        if (
            hasPatientSearch &&
            (
                hasDocumentClassification ||
                hasBadFax
            )
        ) {
            return true;
        }

        return false;
    }

    /**
     * Detect the idle workspace after completing a task.
     *
     * This exact text was present in the provided workspace HTML.
     */
    function isIdleWorkspaceOpen() {
        /*
         * An open fax takes priority, even though the application
         * may still report /workspace as its route.
         */
        if (isFaxTaskOpen()) {
            return false;
        }

        const onWorkspaceRoute =
            location.pathname === '/workspace' ||
            location.pathname === '/workspace/';

        if (!onWorkspaceRoute) {
            return false;
        }

        const pageText = getPageText();

        const caughtUpText =
            pageText.includes(
                'all caught up! no pending tasks right now.'
            );

        if (caughtUpText) {
            return true;
        }

        /*
         * Backup idle-workspace combination.
         */
        const hasAssignedWorkflows =
            pageText.includes(
                'your assigned workflows'
            );

        const hasNeedsReview =
            pageText.includes('needs review');

        const hasDoneToday =
            pageText.includes('done today');

        return (
            hasAssignedWorkflows &&
            hasNeedsReview &&
            hasDoneToday
        );
    }

    /**
     * Update the status dot.
     *
     * Green: actively hunting.
     * Red: stopped or paused because a task is open.
     */
    function updateIndicator() {
        const indicator =
            document.getElementById(
                'tennr-hunt-indicator'
            );

        if (!indicator) {
            return;
        }

        const activelyHunting =
            autoModeEnabled &&
            hunting &&
            !taskOpen;

        indicator.classList.toggle(
            'tennr-hunt-running',
            activelyHunting
        );

        indicator.classList.toggle(
            'tennr-hunt-stopped',
            !activelyHunting
        );
    }

    /**
     * Create the status dot.
     *
     * Custom position and size are preserved exactly.
     */
    function createStatusIndicator() {
        if (
            !document.getElementById(
                'tennr-hunt-indicator-style'
            )
        ) {
            const style =
                document.createElement('style');

            style.id =
                'tennr-hunt-indicator-style';

            style.textContent = `

                    border-radius: 50%;
                    z-index: 2147483647;
                    pointer-events: none;
                    box-sizing: border-box;
                    transition:
                        background-color 0.2s ease,
                        box-shadow 0.2s ease;
                }

                #tennr-hunt-indicator.tennr-hunt-running {
                    background-color: #22c55e;

                        0 0 0 2px rgba(34, 197, 94, 0.30),

        let indicator =

            );

        if (!indicator) {
            indicator =

            indicator.id =
                'tennr-hunt-indicator';

            indicator.className =
                'tennr-hunt-stopped';

            indicator.setAttribute(

            );

            document.body.appendChild(indicator);
        }

        updateIndicator();
    }

    /**
     * Return a random duration between 5 and 12 seconds.
     */
    function randomTime() {
        return (
            Math.floor(Math.random() * 7000) +
            5000
        );
    }

    /**
     * Release Ctrl and Enter.
     */
    function releaseCtrlEnter() {
        document.dispatchEvent(
            new KeyboardEvent('keyup', {
                key: 'Enter',
                code: 'Enter',
                ctrlKey: true,
                bubbles: true
            })
        );

        document.dispatchEvent(
            new KeyboardEvent('keyup', {
                key: 'Control',
                code: 'ControlLeft',
                ctrlKey: false,
                bubbles: true
            })
        );
    }

    /**
     * Send one Ctrl+Enter sequence.
     */
    function fireCtrlEnter() {
        if (
            !autoModeEnabled ||
            !hunting ||
            taskOpen
        ) {
            return;
        }

        /*
         * Final protection immediately before dispatch.
         */
        if (isFaxTaskOpen()) {
            pauseForOpenTask();
            return;
        }

        document.dispatchEvent(
            new KeyboardEvent('keydown', {
                key: 'Control',
                code: 'ControlLeft',
                ctrlKey: true,
                bubbles: true
            })
        );

        document.dispatchEvent(
            new KeyboardEvent('keydown', {
                key: 'Enter',
                code: 'Enter',
                ctrlKey: true,
                bubbles: true
            })
        );

        document.dispatchEvent(
            new KeyboardEvent('keyup', {
                key: 'Enter',
                code: 'Enter',
                ctrlKey: true,
                bubbles: true
            })
        );

        document.dispatchEvent(
            new KeyboardEvent('keyup', {
                key: 'Control',
                code: 'ControlLeft',
                ctrlKey: false,
                bubbles: true
            })
        );
    }

    /**
     * Begin repeated Ctrl+Enter sequences.
     */
    function startCtrlSpam() {
        stopCtrlSpam();

        ctrlSpamTimer = setInterval(() => {
            fireCtrlEnter();
        }, 50);
    }

    /**
     * Stop repeated Ctrl+Enter sequences.
     */
    function stopCtrlSpam() {
        if (ctrlSpamTimer !== null) {
            clearInterval(ctrlSpamTimer);
            ctrlSpamTimer = null;
        }
    }

    /**
     * Hold Ctrl+Enter for a duration.
     */
    function holdCtrlEnter(ms) {
        if (
            !autoModeEnabled ||
            !hunting ||
            taskOpen ||
            isFaxTaskOpen()
        ) {
            if (isFaxTaskOpen()) {
                pauseForOpenTask();
            }

            return;
        }

        document.dispatchEvent(
            new KeyboardEvent('keydown', {
                key: 'Control',
                code: 'ControlLeft',
                ctrlKey: true,
                bubbles: true
            })
        );

        document.dispatchEvent(
            new KeyboardEvent('keydown', {
                key: 'Enter',
                code: 'Enter',
                ctrlKey: true,
                bubbles: true
            })
        );

        holdTimeout = setTimeout(() => {
            holdTimeout = null;
            releaseCtrlEnter();
        }, ms);
    }

    /**
     * Clear hunt-operation timers.
     */
    function clearHuntTimers() {
        stopCtrlSpam();

        if (cycleTimeout !== null) {
            clearTimeout(cycleTimeout);
            cycleTimeout = null;
        }

        if (holdTimeout !== null) {
            clearTimeout(holdTimeout);
            holdTimeout = null;
            releaseCtrlEnter();
        }

        if (reloadTimeout !== null) {
            clearTimeout(reloadTimeout);
            reloadTimeout = null;
        }
    }

    /**
     * Clear the automatic-resume timer.
     */
    function clearResumeTimer() {
        if (resumeTimeout !== null) {
            clearTimeout(resumeTimeout);
            resumeTimeout = null;
        }
    }

    /**
     * Pause immediately when a fax opens.
     */
    function pauseForOpenTask() {
        if (!autoModeEnabled) {
            return;
        }

        const wasPreviouslyOpen = taskOpen;

        taskOpen = true;
        hunting = false;

        taskOpenChecks = 0;
        workspaceChecks = 0;

        clearHuntTimers();
        clearResumeTimer();

        if (!wasPreviouslyOpen) {
            stats.confirmedTasks += 1;
            saveStats();

            console.log(
                '[Tennr Hunt] Fax detected. Hunt paused.'
            );
        }

        updateIndicator();
    }

    /**
     * Resume after the completed workspace is stable.
     */
    function scheduleAutomaticResume() {
        if (
            !autoModeEnabled ||
            !taskOpen ||
            resumeTimeout !== null
        ) {
            return;
        }

        resumeTimeout = setTimeout(() => {
            resumeTimeout = null;

            if (
                !autoModeEnabled ||
                !taskOpen
            ) {
                return;
            }

            /*
             * Confirm the fax did not reopen during the wait.
             */
            if (isFaxTaskOpen()) {
                workspaceChecks = 0;
                return;
            }

            /*
             * Resume only on the verified idle workspace.
             */
            if (!isIdleWorkspaceOpen()) {
                workspaceChecks = 0;
                return;
            }

            taskOpen = false;
            hunting = true;

            lastTaskCandidate = null;
            taskOpenChecks = 0;
            workspaceChecks = 0;
            cycleStartUrl = location.href;

            updateIndicator();

            console.log(
                '[Tennr Hunt] Task completed. Hunt resumed automatically.'
            );

            runCycle();
        }, RESUME_DELAY);
    }

    /**
     * Monitor transitions between hunt and task states.
     */
    function monitorPageState() {
        if (!autoModeEnabled) {
            taskOpenChecks = 0;
            workspaceChecks = 0;
            return;
        }

        const faxVisible = isFaxTaskOpen();

        /*
         * Detect the fax even when the API hook misses it.
         */
        if (!taskOpen && faxVisible) {
            taskOpenChecks += 1;

            if (
                taskOpenChecks >=
                TASK_OPEN_CONFIRMATIONS
            ) {
                pauseForOpenTask();
            }

            return;
        }

        if (!taskOpen) {
            taskOpenChecks = 0;
        }

        /*
         * While paused, wait for the true idle workspace.
         */
        if (taskOpen) {
            if (faxVisible) {
                workspaceChecks = 0;
                clearResumeTimer();
                return;
            }

            if (isIdleWorkspaceOpen()) {
                workspaceChecks += 1;
            } else {
                workspaceChecks = 0;
                clearResumeTimer();
            }

            if (
                workspaceChecks >=
                WORKSPACE_CONFIRMATIONS
            ) {
                scheduleAutomaticResume();
            }
        }
    }

    /**
     * Process an API task candidate.
     */
    function handleTaskCandidate(result) {
        if (!autoModeEnabled || taskOpen) {
            return;
        }

        lastTaskCandidate = result;

        stats.apiCandidates += 1;
        saveStats();

        console.log(
            '[Tennr Hunt] API candidate:',
            result
        );

        monitorPageState();

        requestAnimationFrame(() => {
            monitorPageState();
        });

        setTimeout(() => {
            monitorPageState();
        }, 100);

        setTimeout(() => {
            monitorPageState();
        }, 300);
    }

    /**
     * Inspect a Tennr API response.
     */
    function inspectTaskResponse(data) {
        const result =
            data?.[0]?.result?.data?.json;

        if (
            result === null ||
            result === undefined ||
            result === false
        ) {
            return;
        }

        if (
            Array.isArray(result) &&
            result.length === 0
        ) {
            return;
        }

        if (
            typeof result === 'object' &&
            !Array.isArray(result) &&
            Object.keys(result).length === 0
        ) {
            return;
        }

        handleTaskCandidate(result);
    }

    /**
     * Hook Tennr fetch requests.
     */
    function hookFetch() {
        if (
            typeof window.fetch !== 'function' ||
            window.fetch.__tennrAutoHuntHooked
        ) {
            return;
        }

        const originalFetch = window.fetch;

        async function hookedFetch(...args) {
            const response =
                await originalFetch.apply(this, args);

            try {
                const request = args[0];

                const url =
                    typeof request === 'string'
                        ? request
                        : request?.url || '';

                if (
                    url.includes(
                        'fetchAndAssignNextTask'
                    )
                ) {
                    stats.apiAttempts += 1;
                    saveStats();

                    response
                        .clone()
                        .json()
                        .then((data) => {
                            inspectTaskResponse(data);
                        })
                        .catch((error) => {
                            console.debug(
                                '[Tennr Hunt] Response was not JSON:',
                                error
                            );
                        });
                }
            } catch (error) {
                console.debug(
                    '[Tennr Hunt] Could not inspect request:',
                    error
                );
            }

            return response;
        }

        hookedFetch.__tennrAutoHuntHooked = true;
        window.fetch = hookedFetch;
    }

    /**
     * Observe page changes.
     */
    function startPageObserver() {
        if (taskObserver || !document.body) {
            return;
        }

        let observerCheckPending = false;

        taskObserver = new MutationObserver(() => {
            if (observerCheckPending) {
                return;
            }

            observerCheckPending = true;

            requestAnimationFrame(() => {
                observerCheckPending = false;
                monitorPageState();
            });
        });

        taskObserver.observe(document.body, {
            childList: true,
            subtree: true,
            attributes: true,
            attributeFilter: [
                'class',
                'style',
                'hidden',
                'aria-hidden',
                'title',
                'data-testid',
                'data-page-number',
                'data-validate-scroll-target'
            ]
        });
    }

    /**
     * Start a backup interval monitor.
     */
    function startPageMonitor() {
        if (pageMonitorTimer !== null) {
            return;
        }

        pageMonitorTimer = setInterval(() => {
            monitorPageState();
        }, PAGE_CHECK_INTERVAL);
    }

    /**
     * Handle SPA page navigation.
     */
    function handleNavigationChange() {
        monitorPageState();

        requestAnimationFrame(() => {
            monitorPageState();
        });

        setTimeout(() => {
            monitorPageState();
        }, 100);

        setTimeout(() => {
            monitorPageState();
        }, 300);
    }

    /**
     * Hook Next.js history changes.
     */
    function hookHistoryChanges() {
        if (
            window.__tennrAutoHuntHistoryHooked
        ) {
            return;
        }

        window.__tennrAutoHuntHistoryHooked = true;

        const originalPushState =
            history.pushState;

        const originalReplaceState =
            history.replaceState;

        history.pushState = function (...args) {
            const result =
                originalPushState.apply(
                    this,
                    args
                );

            handleNavigationChange();

            return result;
        };

        history.replaceState = function (...args) {
            const result =
                originalReplaceState.apply(
                    this,
                    args
                );

            handleNavigationChange();

            return result;
        };

        window.addEventListener(
            'popstate',
            handleNavigationChange
        );

        window.addEventListener(
            'hashchange',
            handleNavigationChange
        );
    }

    /**
     * Run one hunt cycle.
     */
    function runCycle() {
        if (
            !autoModeEnabled ||
            !hunting ||
            taskOpen
        ) {
            return;
        }

        /*
         * Do not begin a cycle if a fax is already visible.
         */
        if (isFaxTaskOpen()) {
            pauseForOpenTask();
            return;
        }

        cycleStartUrl = location.href;

        stats.retries += 1;
        saveStats();

        const spamTime = randomTime();

        updateIndicator();
        startCtrlSpam();

        cycleTimeout = setTimeout(() => {
            cycleTimeout = null;

            monitorPageState();

            if (
                !autoModeEnabled ||
                !hunting ||
                taskOpen
            ) {
                stopCtrlSpam();
                return;
            }

            stopCtrlSpam();

            if (isFaxTaskOpen()) {
                pauseForOpenTask();
                return;
            }

            const holdTime = randomTime();

            holdCtrlEnter(holdTime);

            reloadTimeout = setTimeout(() => {
                reloadTimeout = null;

                monitorPageState();

                if (
                    !autoModeEnabled ||
                    !hunting ||
                    taskOpen
                ) {
                    return;
                }

                /*
                 * Final direct check before reload.
                 */
                if (isFaxTaskOpen()) {
                    pauseForOpenTask();
                    return;
                }

                location.reload();
            }, holdTime + 100);
        }, spamTime);
    }

    /**
     * Start or arm automatic hunting.
     */
    function startHunt() {
        autoModeEnabled = true;

        localStorage.setItem(
            AUTO_HUNT_KEY,
            'on'
        );

        clearHuntTimers();
        clearResumeTimer();

        /*
         * F8 while a fax is open arms auto-resume,
         * but does not send Ctrl+Enter.
         */
        if (isFaxTaskOpen()) {
            taskOpen = true;
            hunting = false;

            taskOpenChecks = 0;
            workspaceChecks = 0;

            updateIndicator();

            console.log(
                '[Tennr Hunt] Auto-resume armed while fax is open.'
            );

            return;
        }

        if (hunting && !taskOpen) {
            updateIndicator();
            return;
        }

        taskOpen = false;
        hunting = true;

        lastTaskCandidate = null;
        taskOpenChecks = 0;
        workspaceChecks = 0;
        cycleStartUrl = location.href;

        updateIndicator();

        console.log(
            '[Tennr Hunt] Hunt started with F8.'
        );

        runCycle();
    }

    /**
     * Stop and completely disarm hunting.
     */
    function stopHunt() {
        autoModeEnabled = false;
        hunting = false;
        taskOpen = false;

        lastTaskCandidate = null;
        taskOpenChecks = 0;
        workspaceChecks = 0;

        localStorage.setItem(
            AUTO_HUNT_KEY,
            'off'
        );

        clearHuntTimers();
        clearResumeTimer();

        updateIndicator();

        console.log(
            '[Tennr Hunt] Hunt stopped with F9.'
        );
    }

    /**
     * Check whether the shortcut occurred in an editable field.
     */
    function isTypingTarget(element) {
        if (!(element instanceof Element)) {
            return false;
        }

        return Boolean(
            element.closest(
                'input, textarea, select, [contenteditable="true"]'
            )
        );
    }

    /**
     * Install F8 and F9.
     */
    function installKeyboardShortcuts() {
        if (
            window.__tennrAutoHuntShortcutsInstalled
        ) {
            return;
        }

        window.__tennrAutoHuntShortcutsInstalled = true;

        window.addEventListener(
            'keydown',
            (event) => {
                if (event.repeat) {
                    return;
                }

                if (isTypingTarget(event.target)) {
                    return;
                }

                if (event.key === START_KEY) {
                    event.preventDefault();
                    event.stopPropagation();

                    startHunt();
                    return;
                }

                if (event.key === STOP_KEY) {
                    event.preventDefault();
                    event.stopPropagation();

                    stopHunt();
                }
            },
            true
        );
    }

    /**
     * Restore the saved state after reload.
     */
    function restoreSavedState() {
        if (!autoModeEnabled) {
            hunting = false;
            taskOpen = false;
            updateIndicator();
            return;
        }

        setTimeout(() => {
            if (!autoModeEnabled) {
                return;
            }

            if (isFaxTaskOpen()) {
                taskOpen = true;
                hunting = false;

                updateIndicator();

                console.log(
                    '[Tennr Hunt] Open fax detected after reload.'
                );

                return;
            }

            taskOpen = false;
            hunting = true;
            cycleStartUrl = location.href;

            updateIndicator();
            runCycle();
        }, 1000);
    }

    /**
     * Initialize.
     */
    function initializeTennrHunt() {
        if (!document.head || !document.body) {
            return false;
        }

        try {
            createStatusIndicator();
            installKeyboardShortcuts();
            hookFetch();
            hookHistoryChanges();
            startPageObserver();
            startPageMonitor();
            restoreSavedState();
        } catch (error) {
            console.error(
                '[Tennr Hunt] Initialization failed:',
                error
            );

            return false;
        }

        return true;
    }

    if (!initializeTennrHunt()) {
        const loader = setInterval(() => {
            if (initializeTennrHunt()) {
                clearInterval(loader);
            }
        }, 100);
    }
})();
